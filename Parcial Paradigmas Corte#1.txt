public class Apostador {
   private String nombre;
    private int id;
    private int edad;
    private String correo;
    private String celular;
    private double totalApostado;
    private double totalGanado;
    private String juegoFav;
    private String frecuencia;
    private String estado;
    
    public Apostador() {
        nombre = "Sin nombre";
        id = 0;
        edad = 18;
        correo = "No tiene";
        celular = "0000000000";
        totalApostado = 0;
        totalGanado = 0;
        juegoFav = "Ninguno";
        frecuencia = "ocasional";
        estado = "activa";
    }

    public Apostador(String nombre, int id, int edad, String correo, String celular, double totalApostado, double totalGanado, String juegoFav, String frecuencia, String estado) {
        this.nombre = nombre;
        this.id = id;
        this.edad = edad;
        this.correo = correo;
        this.celular = celular;
        this.totalApostado = totalApostado;
        this.totalGanado = totalGanado;
        this.juegoFav = juegoFav;
        this.frecuencia = frecuencia;
        this.estado = estado;
    }
    
    public Apostador(String nombre, int id, int edad, String correo, String celular) {
        this(nombre, id, edad, correo, celular, 0, 0, "Ninguno", "ocasional", "activa");
    }
    
    public void mostrarInfo() {
        System.out.println("ID: " + id + ", Nombre: " + nombre + ", Correo: " + correo + ", Tel: " + celular);
    }
    
    public void cambiarDatos(String nuevoCorreo, String nuevoCelular) {
        correo = nuevoCorreo;
        celular = nuevoCelular;
    }
    
    public boolean esMayor() {
        return edad >= 18;
    }
    
    public void registrarApuesta(double monto) {
        totalApostado += monto;
    }
    
    public double calcularGanancia() {
        return totalGanado - totalApostado;
    }
    
    public void cambiarEstado() {
        estado = estado.equals("activa") ? "suspendida" : "activa";
    }
    
    public int validarJuego() {
        if (juegoFav.equalsIgnoreCase("loteria") || juegoFav.equalsIgnoreCase("ruleta")) return 3;
        if (juegoFav.equalsIgnoreCase("bingo")) return 1;
        return 0;
    }
    
    public void mensajeFrecuencia() {
        if (frecuencia.equalsIgnoreCase("diaria")) {
            System.out.println("Debes controlarte");
        } else if (frecuencia.equalsIgnoreCase("semanal")) {
            System.out.println("Ten cuidado con no aumentar");
        } else {
            System.out.println("Sigue así");
        }
    }
}
