public class Obra {

    private String id;
    private String titulo;
    private String Genero;
    private String fechaPrestamo;
    private String fechaDevolucion;
    private boolean Disponible;
    private int yearPublicacion;
    private double precio;
    private int diasMora;
    private double valorMulta;
    private int calcularMora;
    private boolean actualizarDatos;

    public Obra() {
        this.id = "0000";
        this.titulo = "Predeterminado";
        this.Genero = "predeterminado";
        this.fechaPrestamo = "00/00/0000";
        this.fechaDevolucion = "00/00/0000";
        this.Disponible = true;
        this.yearPublicacion = 2000;
        this.precio = 0.0;
        this.actualizarDatos = false;
        this.diasMora = 0;
    }

    public Obra(String id, String titulo, String genero, String fechaPrestamo, String fechaDevolucion, boolean disponible, int yearPublicacion, double precio, boolean actualizarDatos) {
        this.id = id;
        this.titulo = titulo;
        this.Genero = genero;
        this.fechaPrestamo = fechaPrestamo;
        this.fechaDevolucion = fechaDevolucion;
        this.Disponible = disponible;
        this.yearPublicacion = yearPublicacion;
        this.precio = precio;
        this.actualizarDatos = actualizarDatos;
        this.valorMulta = calcularMulta();
        this.diasMora = calcularMora();

    }

    public int calcularMora() {
        if (getFechaPrestamo() == null || getFechaDevolucion() == null) {
            return 0;
        }
        return 5;
    }

    public int calcularMulta() {
        return getDiasMora() * 1000;
    }

    public void actualizarDatos() {
        this.setDisponible(!this.isDisponible());
    }

    /**
     * @return the id
     */
    public String getId() {
        return id;
    }

    /**
     * @param id the id to set
     */
    public void setId(String id) {
        this.id = id;
    }

    /**
     * @return the titulo
     */
    public String getTitulo() {
        return titulo;
    }

    /**
     * @param titulo the titulo to set
     */
    public void setTitulo(String titulo) {
        this.titulo = titulo;
    }

    /**
     * @return the Genero
     */
    public String getGenero() {
        return Genero;
    }

    /**
     * @param Genero the Genero to set
     */
    public void setGenero(String Genero) {
        this.Genero = Genero;
    }

    /**
     * @return the fechaPrestamo
     */
    public String getFechaPrestamo() {
        return fechaPrestamo;
    }

    /**
     * @param fechaPrestamo the fechaPrestamo to set
     */
    public void setFechaPrestamo(String fechaPrestamo) {
        this.fechaPrestamo = fechaPrestamo;
    }

    /**
     * @return the fechaDevolucion
     */
    public String getFechaDevolucion() {
        return fechaDevolucion;
    }

    /**
     * @param fechaDevolucion the fechaDevolucion to set
     */
    public void setFechaDevolucion(String fechaDevolucion) {
        this.fechaDevolucion = fechaDevolucion;
    }

    /**
     * @return the Disponible
     */
    public boolean isDisponible() {
        return Disponible;
    }

    /**
     * @param Disponible the Disponible to set
     */
    public void setDisponible(boolean Disponible) {
        this.Disponible = Disponible;
    }

    /**
     * @return the yearPublicacion
     */
    public int getYearPublicacion() {
        return yearPublicacion;
    }

    /**
     * @param yearPublicacion the yearPublicacion to set
     */
    public void setYearPublicacion(int yearPublicacion) {
        this.yearPublicacion = yearPublicacion;
    }

    /**
     * @return the precio
     */
    public double getPrecio() {
        return precio;
    }

    /**
     * @param precio the precio to set
     */
    public void setPrecio(double precio) {
        this.precio = precio;
    }

    /**
     * @return the diasMora
     */
    public int getDiasMora() {
        return diasMora;
    }

    /**
     * @param diasMora the diasMora to set
     */
    public void setDiasMora(int diasMora) {
        this.diasMora = diasMora;
    }

    /**
     * @return the valorMulta
     */
    public double getValorMulta() {
        return valorMulta;
    }

    /**
     * @param valorMulta the valorMulta to set
     */
    public void setValorMulta(double valorMulta) {
        this.valorMulta = valorMulta;
    }

    /**
     * @return the calcularMora
     */
    public int getCalcularMora() {
        return calcularMora;
    }

    /**
     * @param calcularMora the calcularMora to set
     */
    public void setCalcularMora(int calcularMora) {
        this.calcularMora = calcularMora;
    }

    /**
     * @return the actualizarDatos
     */
    public boolean isActualizarDatos() {
        return actualizarDatos;
    }

    /**
     * @param actualizarDatos the actualizarDatos to set
     */
    public void setActualizarDatos(boolean actualizarDatos) {
        this.actualizarDatos = actualizarDatos;
    }
    
    public void mostrarInformacion () {
        System.out.println("ID: " + this.id);
        System.out.println("Titulo: " + this.titulo);
        System.out.println("Genero: " + this.Genero);
        System.out.println("Fecha de prestamo: " + this.fechaPrestamo);
        System.out.println("Fecha de devolucion: " + this.fechaDevolucion);
        System.out.println("Disponible: " + this.Disponible);
        System.out.println("Ano de publicacion: " + this.yearPublicacion);
        System.out.println("Precio: " + this.precio);
        System.out.println("Dias de mora: " + this.diasMora);
        System.out.println("Valor de la multa: " + this.valorMulta);
    }
}
import java.util.Scanner;

public class ExamenPoliherencia {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        Obra obra1 = new Obra();

        System.out.println("Datos de la obra:");
        obra1.mostrarInformacion();

        System.out.println("\nActualizar disponibilidad:");
        obra1.actualizarDatos();

        System.out.println("\nIngrese los datos de la nueva obra:");
        System.out.print("ID: ");
        String id = sc.nextLine();
        System.out.print("Titulo: ");
        String titulo = sc.nextLine();
        System.out.print("Genero: ");
        String genero = sc.nextLine();
        System.out.print("Fecha de prestamo (dd/mm/aaaa): ");
        String fechaPrestamo = sc.nextLine();
        System.out.print("Fecha de devolucion (dd/mm/aaaa): ");
        String fechaDevolucion = sc.nextLine();
        System.out.print("Ano de publicacion: ");
        int yearPublicacion = sc.nextInt();
        System.out.print("Precio: ");
        double precio = sc.nextDouble();
        System.out.print(" Esta disponible : ");
        boolean disponible = sc.nextBoolean();

        Obra obra2 = new Obra(id, titulo, genero, fechaPrestamo, fechaDevolucion, disponible, yearPublicacion, precio, false);
        System.out.println("\nDatos de la nueva obra:");
        obra2.mostrarInformacion();

    }

}
